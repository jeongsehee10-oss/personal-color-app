# 3. 진단 결과 판정 및 출력
if submitted:
    score_w = sum(1 for q in [q1, q2, q3, q4, q5, q6, q7, q8, q9, q10] if q.startswith("2"))
    score_c = 10 - score_w

    score_s1 = sum(1 for q in [q11, q12, q13, q14, q15, q16, q17, q18, q19, q20] if q.startswith("1"))
    score_s2 = 10 - score_s1

    score_tl = 10 - score_th

    base_res = "웜" if score_w >= score_c else "쿨"

    if base_res == "웜":
        if score_s1 >= score_s2:
            tone_name = "봄웜 라이트" if score_tl >= score_th else "봄웜 브라이트"
        else:
            tone_name = "가을웜 뮤트" if score_th >= score_tl else "가을웜 딥"
    else:
        if score_s1 >= score_s2:
            tone_name = "여름 쿨톤 라이트" if score_tl >= score_th else "여름 쿨톤 뮤트"
        else:
            tone_name = "겨울 쿨톤 트루" if score_tl >= score_th else "겨울 쿨톤 브라이트"
